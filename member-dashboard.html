/* ============================================ CORTEX MEMBER DASHBOARD — Content + Access Grants UI Reads from backend APIs to display member content ============================================ */ document.addEventListener('DOMContentLoaded', async () => { const JWT_TOKEN = localStorage.getItem('cortex_jwt'); const userEmail = getUserEmailFromToken(); if (!JWT_TOKEN || !userEmail) { document.body.innerHTML = `

Cortex Member Dashboard

Please log in to access member content.

`; return; } try { // Fetch member data + access grants const meResponse = await fetch('/api/auth/me', { headers: { 'Authorization': `Bearer ${JWT_TOKEN}` } }); const meData = await meResponse.json(); // Fetch content payload (podcast registry + access grants) const contentResponse = await fetch(`/api/member-payload?email=${encodeURIComponent(userEmail)}`, { headers: { 'Authorization': `Bearer ${JWT_TOKEN}` } }); const contentData = contentResponse.status === 200 ? await contentResponse.json() : {}; renderMemberDashboard(meData, contentData); } catch (err) { console.error('Error loading member dashboard:', err); document.body.innerHTML = `

Error Loading Dashboard

${err.message}

`; } }); function getUserEmailFromToken() { const token = localStorage.getItem('cortex_jwt'); if (!token) return null; try { const parts = token.split('.'); const payload = JSON.parse(atob(parts[1])); return payload.email; } catch { return null; } } function renderMemberDashboard(userData, contentData) { const grants = (userData && userData.grants) || (contentData && contentData.access_grants) || []; const content = (contentData && contentData.content) || []; const user = userData && userData.user ? userData.user : {}; let html = `

🎙️ Cortex Intelligence Nexus — Member Dashboard

${user.name || 'Member'} (${user.email})

Welcome to your exclusive content library. Stream podcasts, read resources, and access member-only insights.

✅ Your Access Grants

${grants && grants.length > 0 ? ` ${grants.map((grant, i) => { const expiresDate = new Date(grant.expires_at || grant.granted_at); const daysLeft = Math.max(0, Math.ceil((expiresDate - new Date()) / (1000 * 60 * 60 * 24))); return `
${grant.product_name || grant.product}
✓ Active
${daysLeft > 0 ? `Expires in ${daysLeft} days` : 'Expires today'}
`; }).join('')} ` : `

No active access grants yet.

Subscribe Now — ₦22,000

`}

🎬 Member Content Library

${content && content.length > 0 ? `
${content.map((item, i) => `

${item.title || 'Episode ' + (i + 1)}

Type: ${item.type === 'podcast' ? '🎙️ Podcast' : item.type}

Added: ${new Date(item.created_at).toLocaleDateString()}

${item.status === 'unlocked' ? ` 📖 View Full Content ` : `
🔒 Locked
`}
`).join('')}
` : `

No member content available yet. Check back soon!

`}

📢 Upgrade Your Membership

Want access to more content? Upgrade your membership or renew your current plan.

Explore Offers
`; document.body.innerHTML = html; }
